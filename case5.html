javascript:(function(){
  if(window.ptT) clearInterval(window.ptT);
  if(window.scT) clearInterval(window.scT);

  var old=document.getElementById('ps_s');
  if(old) old.remove();

  // 하이라이팅을 위한 고시인성 CSS 스타일 동적 인젝션
  var s=document.createElement('style');
  s.id='ps_s';
  s.innerHTML='.t-hi td{background:#e8f5e9!important;border-top:2px solid #4caf50!important;border-bottom:2px solid #4caf50!important;font-weight:bold!important;color:#1b5e20!important;transition:all 0.3s;}.t-hi-pending td{background:#fff3e0!important;border-top:2px solid #ff9800!important;border-bottom:2px solid #ff9800!important;font-weight:bold!important;color:#b45309!important;transition:all 0.3s;}';
  document.head.appendChild(s);

  function h(){
    var now=new Date(),
        yr=now.getFullYear(),
        hiList=[],
        nxList=[],
        mD=Infinity;

    // 전체 tr 순회하며 시각 비교
    document.querySelectorAll('tr').forEach(function(r){
      r.classList.remove('t-hi');
      r.classList.remove('t-hi-pending');
      var tds=r.querySelectorAll('td');
      if(tds.length<5) return;
      var dt=tds[3].innerText.trim(),
          tt=tds[4].innerText.trim();
      if(!dt||!tt) return;
      if(!dt.match(/^\d{2}-\d{2}$/)||!tt.match(/^\d{2}:\d{2}$/)) return;
      var p=dt.split('-'),t=tt.split(':'),shipTime=new Date(yr,p[0]-1,p[1],t[0],t[1]),
          df=shipTime-now;
      if(df>=0&&df<=3600000){
        hiList.push(r);
      }else if(df>0){
        if(df<mD-60000){
          mD=df;
          nxList=[r];
        }else if(df<=mD+60000){
          if(df<mD) mD=df;
          nxList.push(r);
        }
      }
    });

    var targets=hiList.length?hiList:nxList;
    targets.forEach(function(r){
      var tds=r.querySelectorAll('td');
      var pilot=tds[2].innerText.trim();
      if(pilot){
        r.classList.add('t-hi');
      }else{
        r.classList.add('t-hi-pending');
      }
    });
  }

  // 타겟 행을 화면 정중앙으로 부드럽게 스크롤
  function sC(){
    var t=document.querySelector('.t-hi,.t-hi-pending');
    if(t) t.scrollIntoView({behavior:'smooth', block:'center'});
  }

  h();
  setTimeout(sC, 100);
  window.ptT=setInterval(h, 180000);
  window.scT=setInterval(sC, 180000);
})();
