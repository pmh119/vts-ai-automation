/**
 * 스프레드시트 인스턴스 가져오기
 */
function getSpreadsheet() {
  if (SPREADSHEET_ID && SPREADSHEET_ID.trim() !== '') {
    return SpreadsheetApp.openById(SPREADSHEET_ID);
  }
  return SpreadsheetApp.getActiveSpreadsheet();
}

/**
 * 게시판용 시트(VTS_Board) 확인 및 초기화
 */
function checkAndInitSheet() {
  var ss = getSpreadsheet();
  var sheet = ss.getSheetByName('VTS_Board');
  if (!sheet) {
    sheet = ss.insertSheet('VTS_Board');
    // 헤더 구조 설정: ID, 작성자, 제목, 내용, 작성일, 비밀번호, 타임스탬프
    sheet.getRange(1, 1, 1, 7).setValues([['ID', '작성자', '제목', '내용', '작성일', '비밀번호', '등록시각']]);
    
    // 초기 샘플 데이터 작성
    var sampleDate = new Date();
    var formatDate = function(d) {
      var y = d.getFullYear();
      var m = String(d.getMonth() + 1).padStart(2, '0');
      var dd = String(d.getDate()).padStart(2, '0');
      return y + '-' + m + '-' + dd;
    };
    sheet.appendRow([
      'sample-1',
      '관리자',
      '평택 VTS 자유게시판 개설을 환영합니다!',
      'VTS 현장 업무 자동화 사례에 대한 자유로운 소감, 개량 아이디어, 버그 피드백을 나눠주세요.',
      formatDate(sampleDate),
      '1234',
      sampleDate
    ]);
  }
  return sheet;
}

/**
 * GET 요청 처리: 전체 게시글 목록 가져오기
 */
function doGet(e) {
  try {
    var sheet = checkAndInitSheet();
    var values = sheet.getDataRange().getValues();
    var posts = [];
    
    if (values.length > 1) {
      var headers = values[0];
      var idIdx = headers.indexOf('ID');
      var authorIdx = headers.indexOf('작성자');
      var titleIdx = headers.indexOf('제목');
      var contentIdx = headers.indexOf('내용');
      var dateIdx = headers.indexOf('작성일');
      var pwIdx = headers.indexOf('비밀번호'); // 필요 시 검증용 (클라이언트에는 해시 또는 비교용으로만 전달)
      
      for (var i = 1; i < values.length; i++) {
        var row = values[i];
        
        // 작성일 날짜 포맷팅
        var dateVal = dateIdx >= 0 ? row[dateIdx] : '';
        var formattedDate = '';
        if (dateVal instanceof Date) {
          formattedDate = dateVal.getFullYear() + '-' + String(dateVal.getMonth()+1).padStart(2,'0') + '-' + String(dateVal.getDate()).padStart(2,'0');
        } else if (dateVal) {
          formattedDate = dateVal.toString().split('T')[0];
        }
        
        posts.push({
          id: idIdx >= 0 ? row[idIdx].toString() : '',
          author: authorIdx >= 0 ? row[authorIdx].toString() : '',
          title: titleIdx >= 0 ? row[titleIdx].toString() : '',
          content: contentIdx >= 0 ? row[contentIdx].toString() : '',
          date: formattedDate,
          hasPassword: pwIdx >= 0 && row[pwIdx].toString().trim() !== ''
        });
      }
    }
    
    // 최근 글이 가장 위에 보이도록 내림차순 정렬
    posts.reverse();
    
    return ContentService.createTextOutput(JSON.stringify({ success: true, posts: posts }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * POST 요청 처리: 게시글 신규 등록 및 삭제
 */
function doPost(e) {
  try {
    var sheet = checkAndInitSheet();
    
    // JSON 페이로드 파싱
    var params = {};
    if (e && e.postData && e.postData.contents) {
      params = JSON.parse(e.postData.contents);
    } else if (e && e.parameter) {
      params = e.parameter;
    }
    
    var action = params.action || 'add';
    
    // 1. 게시글 삭제 처리
    if (action === 'delete') {
      var idToDelete = params.id;
      var passwordInput = params.password;
      
      if (!idToDelete || !passwordInput) {
        return ContentService.createTextOutput(JSON.stringify({ success: false, error: 'ID와 비밀번호를 모두 입력해야 합니다.' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
      
      var values = sheet.getDataRange().getValues();
      var headers = values[0];
      var idIdx = headers.indexOf('ID');
      var pwIdx = headers.indexOf('비밀번호');
      
      var foundRow = -1;
      for (var i = 1; i < values.length; i++) {
        if (idIdx >= 0 && values[i][idIdx].toString() === idToDelete) {
          // 비밀번호 확인
          if (pwIdx >= 0 && values[i][pwIdx].toString() === passwordInput.toString()) {
            foundRow = i + 1; // 1-indexed row number
            break;
          } else {
            return ContentService.createTextOutput(JSON.stringify({ success: false, error: '비밀번호가 일치하지 않습니다.' }))
              .setMimeType(ContentService.MimeType.JSON);
          }
        }
      }
      
      if (foundRow > 0) {
        sheet.deleteRow(foundRow);
        return ContentService.createTextOutput(JSON.stringify({ success: true, message: '글이 성공적으로 삭제되었습니다.' }))
          .setMimeType(ContentService.MimeType.JSON);
      } else {
        return ContentService.createTextOutput(JSON.stringify({ success: false, error: '대상 글을 찾을 수 없거나 이미 삭제되었습니다.' }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    
    // 2. 게시글 추가 처리 (기본값)
    var author = params.author;
    var title = params.title;
    var content = params.content;
    var password = params.password;
    
    if (!author || !title || !content || !password) {
      return ContentService.createTextOutput(JSON.stringify({ success: false, error: '작성자, 제목, 내용, 비밀번호를 모두 채워야 합니다.' }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var timestamp = new Date();
    var formatDate = function(d) {
      var y = d.getFullYear();
      var m = String(d.getMonth() + 1).padStart(2, '0');
      var dd = String(d.getDate()).padStart(2, '0');
      return y + '-' + m + '-' + dd;
    };
    
    var postId = 'post-' + timestamp.getTime() + '-' + Math.floor(Math.random() * 1000);
    
    sheet.appendRow([
      postId,
      author.toString().trim(),
      title.toString().trim(),
      content.toString().trim(),
      formatDate(timestamp),
      password.toString().trim(),
      timestamp
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ success: true, message: '글이 등록되었습니다.' }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
